cmake_minimum_required(VERSION 3.17)
project(InvFinder VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(Z3_ROOT "" CACHE PATH "Prefix containing a matching Z3 include/ and lib/")
if(Z3_ROOT)
  find_path(Z3_INCLUDE_DIR z3++.h HINTS "${Z3_ROOT}/include" NO_DEFAULT_PATH)
  find_library(Z3_LIBRARY NAMES z3 HINTS "${Z3_ROOT}/lib" "${Z3_ROOT}/lib64" NO_DEFAULT_PATH)
else()
  find_package(PkgConfig QUIET)
  if(PkgConfig_FOUND)
    pkg_check_modules(PC_Z3 QUIET z3)
  endif()
  find_path(Z3_INCLUDE_DIR z3++.h HINTS ${PC_Z3_INCLUDE_DIRS})
  get_filename_component(_z3_prefix "${Z3_INCLUDE_DIR}" DIRECTORY)
  find_library(Z3_LIBRARY NAMES z3 HINTS ${PC_Z3_LIBRARY_DIRS}
    "${_z3_prefix}/lib" "${_z3_prefix}/lib64")
endif()
if(NOT Z3_INCLUDE_DIR OR NOT Z3_LIBRARY)
  message(FATAL_ERROR "Install the Z3 C++ development files or set -DZ3_ROOT=/path/to/z3")
endif()
message(STATUS "Z3 headers: ${Z3_INCLUDE_DIR}")
message(STATUS "Z3 library: ${Z3_LIBRARY}")
add_library(Z3::Z3 UNKNOWN IMPORTED)
set_target_properties(Z3::Z3 PROPERTIES IMPORTED_LOCATION "${Z3_LIBRARY}"
  INTERFACE_INCLUDE_DIRECTORIES "${Z3_INCLUDE_DIR}")

add_library(invfinder src/transition_system.cpp src/invariant.cpp src/k_induction.cpp)
target_include_directories(invfinder PUBLIC include)
target_link_libraries(invfinder PUBLIC Z3::Z3)
target_compile_options(invfinder PRIVATE $<$<CXX_COMPILER_ID:GNU,Clang>:-Wall;-Wextra;-Wpedantic>)
add_executable(bestInv src/bestinv_main.cpp)
target_link_libraries(bestInv PRIVATE invfinder)
add_executable(kInductor src/kinductor_main.cpp)
target_link_libraries(kInductor PRIVATE invfinder)
add_executable(native_example examples/native.cpp)
target_link_libraries(native_example PRIVATE invfinder)

include(CTest)
if(BUILD_TESTING)
  add_executable(invfinder_tests tests/core_tests.cpp)
  target_link_libraries(invfinder_tests PRIVATE invfinder)
  add_test(NAME invfinder_core COMMAND invfinder_tests)
  set_tests_properties(invfinder_core PROPERTIES TIMEOUT 180)
  add_executable(parser_tests tests/test_parser.cpp)
  target_link_libraries(parser_tests PRIVATE invfinder)
  add_test(NAME invfinder_parser COMMAND parser_tests ${CMAKE_CURRENT_SOURCE_DIR}/examples)
  add_executable(iteration_tests tests/iteration_tests.cpp)
  target_link_libraries(iteration_tests PRIVATE invfinder)
  add_test(NAME invfinder_iterations COMMAND iteration_tests)
  set_tests_properties(invfinder_iterations PROPERTIES TIMEOUT 180)
  find_package(Python3 3.9 COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME invfinder_cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_tests.py
      $<TARGET_FILE:bestInv> $<TARGET_FILE:kInductor> ${CMAKE_CURRENT_SOURCE_DIR})
    set_tests_properties(invfinder_cli PROPERTIES TIMEOUT 90)
    add_test(NAME invfinder_analysis COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/scripts/test_analysis.py)
    set_tests_properties(invfinder_analysis PROPERTIES TIMEOUT 30)
  endif()
endif()
